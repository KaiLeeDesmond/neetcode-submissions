class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        l = 0
        length = 0
        tes = set()
        for r in range(len(s)):
            while s[r] in tes: #we always check with r since that's the loop variable.
                tes.remove(s[l])
                l+=1
            tes.add(s[r])
            length = max(length, r-l+1)

        return length
